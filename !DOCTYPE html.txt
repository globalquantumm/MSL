    <script>
        // 숫자 애니메이션을 위한 헬퍼 함수
        function animateValue(obj, start, end, duration) {
            let startTimestamp = null;
            const step = (timestamp) => {
                if (!startTimestamp) startTimestamp = timestamp;
                const progress = Math.min((timestamp - startTimestamp) / duration, 1);
                
                let currentValue = start + (end - start) * progress;
                
                if (obj.id === 'participants-count') {
                    obj.innerHTML = Math.floor(currentValue).toLocaleString();
                    if (progress === 1) {
                        obj.innerHTML = "3,150+";
                    }
                } 
                else if (obj.id === 'current-price') {
                    obj.innerHTML = `$${currentValue.toFixed(2)}`;
                }
                
                if (progress < 1) {
                    window.requestAnimationFrame(step);
                }
            };
            window.requestAnimationFrame(step);
        }

        const playButton = document.getElementById('playButton');
        const participantsElement = document.getElementById('participants-count');
        const priceElement = document.getElementById('current-price');

        playButton.addEventListener('click', () => {
            playButton.disabled = true;
            playButton.textContent = 'Animating...'; 
            playButton.classList.add('opacity-50', 'cursor-not-allowed');

            participantsElement.innerHTML = "2,789";
            priceElement.innerHTML = "$6.79";

            animateValue(participantsElement, 2789, 3150, 5000); 
            animateValue(priceElement, 6.79, 4.90, 5000);   

            setTimeout(() => {
                playButton.disabled = false;
                playButton.textContent = 'Play Again'; 
                playButton.classList.remove('opacity-50', 'cursor-not-allowed');
            }, 5000);
        });
    </script>
